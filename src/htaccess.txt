# MaxCrit: clean URLs. Pages are stored as name.html and served at /name.
Options -MultiViews
DirectorySlash Off
DirectoryIndex index.html
ErrorDocument 404 /404.html

RewriteEngine On

# HTTPS and no www
RewriteCond %{HTTPS} !=on [OR]
RewriteCond %{HTTP_HOST} ^www\. [NC]
RewriteRule ^ https://maxcrit.com%{REQUEST_URI} [R=301,L]

# Old addresses: /index.html -> /, /name.html -> /name (permanent)
RewriteCond %{THE_REQUEST} \s/+(.*?/)?index\.html[\s?] [NC]
RewriteRule ^ /%1 [R=301,L,NE]
RewriteCond %{THE_REQUEST} \s/+([^?\s]+?)\.html[\s?] [NC]
RewriteRule ^ /%1 [R=301,L,NE]

# Trailing slash on a page address: /hub/ -> /hub
RewriteCond %{DOCUMENT_ROOT}/$1.html -f
RewriteRule ^(.+)/$ /$1 [R=301,L]

# Serve /name from name.html
RewriteCond %{DOCUMENT_ROOT}/$1.html -f
RewriteRule ^(.+)$ $1.html [L]
